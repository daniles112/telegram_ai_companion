    from PySide6.QtWidgets import QWidget, QVBoxLayout, QHBoxLayout
    from GUI_styles_helper import *
    
    class MySettingsPage(QWidget):
        def __init__(self):
            super().__init__()
            
            # Создаем элементы красиво
            title = create_title("Настройки")
            subtitle = create_subtitle("Оформите свой профиль")
            
            name_input = create_input("Введите имя")
            email_input = create_input("Введите email")
            
            save_btn = create_button("Сохранить", style="primary")
            cancel_btn = create_button("Отмена", style="secondary")
            
            # Собираем layout
            layout = QVBoxLayout()
            layout.addWidget(title)
            layout.addWidget(subtitle)
            layout.addWidget(name_input)
            layout.addWidget(email_input)
            
            buttons = QHBoxLayout()
            buttons.addWidget(save_btn)
            buttons.addWidget(cancel_btn)
            buttons.addStretch()
            
            layout.addLayout(buttons)
            layout.addStretch()
            
            self.setLayout(layout)
